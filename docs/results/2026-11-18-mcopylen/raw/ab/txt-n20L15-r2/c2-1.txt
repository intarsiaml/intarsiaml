# lru_cache.py
"""LRU Cache implementation using a doubly linked list and a dictionary."""


class _Node:
    """A node in the doubly linked list."""
    __slots__ = ('key', 'value', 'prev', 'next')

    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """
    LRU Cache built on a doubly linked list and a dict.
    
    Supports:
        - put(key, value): Insert or update a key-value pair.
        - get(key): Retrieve the value for a key, moving it to the front (most recently used).
        - delete(key): Remove a key from the cache.
        - __len__(): Return the number of items in the cache.
    """

    def __init__(self, capacity: int):
        if capacity <= 0:
            raise ValueError("Capacity must be positive")
        self.capacity = capacity
        self.size = 0
        self.cache = {}  # key -> _Node

        # Sentinel nodes for the doubly linked list
        self.head = _Node()  # Most recently used end
        self.tail = _Node()  # Least recently used end
        self.head.next = self.tail
        self.tail.prev = self.head

    def _add_to_front(self, node: _Node):
        """Add a node right after the head (most recently used position)."""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node

    def _remove_node(self, node: _Node):
        """Remove a node from its current position in the linked list."""
        prev_node = node.prev
        next_node = node.next
        prev_node.next = next_node
        next_node.prev = prev_node
        node.prev = None
        node.next = None

    def _move_to_front(self, node: _Node):
        """Move an existing node to the front (most recently used)."""
        self._remove_node(node)
        self._add_to_front(node)

    def _pop_tail(self) -> _Node:
        """Remove and return the node just before the tail (least recently used)."""
        node = self.tail.prev
        self._remove_node(node)
        return node

    def put(self, key, value):
        """Insert or update a key-value pair."""
        if key in self.cache:
            node = self.cache[key]
            node.value = value
            self._move_to_front(node)
        else:
            if self.size >= self.capacity:
                # Evict the least recently used item
                lru_node = self._pop_tail()
                del self.cache[lru_node.key]
                self.size -= 1
            new_node = _Node(key=key, value=value)
            self._add_to_front(new_node)
            self.cache[key] = new_node
            self.size += 1

    def get(self, key):
        """Retrieve the value for a key. Returns -1 if not found."""
        if key not in self.cache:
            return -1
        node = self.cache[key]
        self._move_to_front(node)
        return node.value

    def delete(self, key):
        """Remove a key from the cache. Returns True if deleted, False otherwise."""
        if key not in self.cache:
            return False
        node = self.cache[key]
        self._remove_node(node)
        del self.cache[key]
        self.size
