"""LRU Cache implementation using a doubly linked list and a dictionary."""


class Node:
    """A node in the doubly linked list."""

    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class DoublyLinkedList:
    """Doubly linked list with sentinel head and tail nodes."""

    def __init__(self):
        self.head = Node()  # Sentinel head
        self.tail = Node()  # Sentinel tail
        self.head.next = self.tail
        self.tail.prev = self.head
        self.size = 0

        """Insert new_node after the given node."""
        next_node = node.next
        new_node.prev = node
        new_node.next = next_node
        node.next = new_node
        next_node.prev = new_node

        prev_node = node.prev
        next_node = node.next
        prev_node.next = next_node
