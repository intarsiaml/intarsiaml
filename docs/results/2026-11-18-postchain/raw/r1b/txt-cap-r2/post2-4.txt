class Node:
    """A node in the doubly linked list."""
    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """LRU Cache implemented with a doubly linked list and a dict."""
    
    def __init__(self, capacity: int):
        if capacity <= 0:
            raise ValueError("Capacity must be positive")
        self.capacity = capacity
        self.size = 0
        # Dict mapping key to Node
        self.cache = {}
        # Dummy head and tail for the doubly linked list
        self.head = Node()  # head is the most recent end
        self.tail = Node()  # tail is the least recent end
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _add_to_head(self, node: Node):
        """Add a node right after the head (most recent position)."""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node

        """Remove a node from its current position in the linked list."""
