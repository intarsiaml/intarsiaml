#!/usr/bin/env bash
# Production launcher for spark1's flight/guidance/capcom roles (missioncntrl-guidance), recreated
# from `docker inspect` on 2026-10-22. Only change from the hand-made container: the KV pool is
# sized with --kv-cache-memory-bytes instead of --gpu-memory-utilization 0.50 (which gave a
# 35 GiB pool, 80x the 2 x 32768 tokens --max-num-seqs needs). --gpu-memory-utilization stays: vLLM
# still checks free >= total*util at startup (default 0.92 fails), but no longer sizes the pool. 0.28 =
# 34.1 GiB, just above actual use (~23.7 weights+non-torch, ~2 activations, 4 KV, 0.5 graphs); keep it
# close to use so a restart does not need memory fleetq has handed to other jobs.
# Image: missioncntrl/vllm-openai:65b7662d = f733c5dd166c (vLLM nightly 65b7662d, pinned so a pull of
# nightly-aarch64 cannot move it).
set -euo pipefail
KV_BYTES="${KV_BYTES:-4294967296}"
NAME="${NAME:-missioncntrl-guidance}"
docker run -d --name "$NAME" --restart unless-stopped \
  --gpus all --network host --ipc host --shm-size 16g --security-opt label=disable \
  -v /home/nvidia/.cache/huggingface:/root/.cache/huggingface \
  -e HF_HUB_OFFLINE=1 -e VLLM_LOGGING_LEVEL=INFO \
  -e VLLM_NVFP4_GEMM_BACKEND=flashinfer-cutlass -e VLLM_USE_FLASHINFER_MOE_FP4=0 \
  -e VLLM_DISABLED_KERNELS=FlashInferFP8ScaledMMLinearKernel \

  missioncntrl/vllm-openai:65b7662d \

    --host 100.120.88.104 --port 8000 \
    --served-model-name guidance capcom \
    --quantization modelopt_fp4 \
    --max-model-len 32768 --max-num-seqs 2 --max-num-batched-tokens 8192 \
    --kv-cache-dtype fp8 --kv-cache-memory-bytes "$KV_BYTES" \
    --gpu-memory-utilization 0.28 \
