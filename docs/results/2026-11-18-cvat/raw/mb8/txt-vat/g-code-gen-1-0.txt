```python
# lru_cache.py

class Node:
    """A node in the doubly linked list."""
    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """
    LRU Cache implementation using a doubly linked list and a dictionary.
    
    Supports:
    - put(key, value): Insert or update a key-value pair.
    - get(key): Retrieve the value for a key, moving it to the front (most recently used).
    - delete(key): Remove a key from the cache.
    - __len__(): Return the number of items in the cache.
    """
    
    def __init__(self, capacity: int):
        if capacity <= 0:
            raise ValueError("Capacity must be positive")
        self.capacity = capacity
        self.cache = {}  # key -> Node
        # Sentinel nodes for head and tail
        self.head = Node()  # Most recently used end
        self.tail = Node()  # Least recently used end
        self.head.next = self.tail
        self.tail.prev = self.head
    
    def _add_to_front(self, node: Node):
        """Add a node right after the head (most recently used position)."""
        node.prev = self.head
        node.next = self.head.next
        self.head.next.prev = node
        self.head.next = node
    
    def _remove_node(self, node: Node):
        """Remove a node from the linked list."""
        prev_node = node.prev
        next_node = node.next
        prev_node.next = next_node
        next_node.prev = prev_node
    
    def _move_to_front(self, node: Node):
        """Move an existing node to the front (most recently used)."""
        self._remove_node(node)
        self._add_to_front(node)
    
    def _pop_tail(self) -> Node:
        """Remove and return the node before the tail (least recently used)."""
        node = self.tail.prev
        self._remove_node(node)
        return node
    
    def put(self, key, value):
        """Insert or update a key-value pair."""
        if key in self.cache:
            node = self.cache[key]
            node.value = value
            self._move_to_front(node)
        else:
            if len(self.cache) >= self.capacity:
                # Remove the least recently used item
                lru_node = self._pop_tail()
                del self.cache[lru_node.key]
            new_node = Node(key=key, value=value)
            self.cache[key] = new_node
            self._add_to_front(new_node)
    
    def get(self, key):
        """Retrieve the value for a key. Returns -1 if not found."""
        if key in self.cache:
            node = self.cache[key]
            self._move_to_front(node)
            return node.value
        return -1
    
    def delete(self, key):
        """Remove a key from the cache."""
        if key in self.cache:
            node = self.cache[key]
            self._remove_node(node)
            del self.cache[key]
    
    def __len__(self):
        return len(self.cache)
```
