class _Node:
    """A node in the doubly linked list."""
    __slots__ = ('key', 'value', 'prev', 'next')

    def __init__(self, key=None, value=None):
        self.key = key
        self.value = value
        self.prev = None
        self.next = None


class LRUCache:
    """
    LRU Cache implementation using a doubly linked list and a dictionary.
    
    The doubly linked list maintains the order of usage (most recently used at the head,
    least recently used at the tail). The dictionary provides O(1) lookup by key.
    """

    def __init__(self, capacity: int):
        if capacity <= 0:
            raise ValueError("Capacity must be positive")
        self.capacity = capacity
        self.size = 0

        self.head.next = self.tail
        self.tail.prev = self.head
